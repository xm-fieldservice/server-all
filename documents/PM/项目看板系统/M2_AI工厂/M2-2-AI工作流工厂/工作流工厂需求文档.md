## 一、背景与目标

### 1.1 业务背景

基于M2-1 AI工厂架构底座，我们已具备：
- LangGraph编排引擎
- PostgreSQL/pgvector数据存储
- 1号/2号写库通道
- Finance Agent工作流实践

当前需求：**将单一工作流扩展为可批量生产、可管理、可复用的工作流工厂**

### 1.2 核心目标

| 目标 | 描述 | 优先级 |
|------|------|--------|
| **原生框架** | 定义工作流的通用节点模型、责任人、输入输出 | P0 |
| **快速生产** | 提供工具和规范，快速创建新工作流 | P0 |
| **货架管理** | 模板+实例的分层管理，支持复用和复制 | P1 |
| **知识积累** | 流程数据自动归档，形成知识库素材 | P2 |

```python
class WorkflowNode(BaseModel):
    """工作流节点定义"""
    
    # ==================== 基础属性 ====================
    node_id: str                              # 节点唯一标识 (e.g., "node_001")
    node_name: str                             # 节点名称 (e.g., "直属经理审批")
    node_type: Literal[
        "manual",       # 人工节点
        "system",       # 系统节点
        "decision",     # 决策节点
        "llm",          # LLM节点
        "subflow"       # 子流程节点
    ]
    description: Optional[str] = None          # 节点描述
    version: str = "1.0.0"                    # 版本号
    
    # ==================== 责任人体系 ====================
    assignee_type: Literal[
        "user",         # 固定用户
        "role",         # 角色 (如"部门经理")
        "department",   # 部门 (如"所属部门")
        "dynamic",      # 动态计算 (如"申请人直属上级")
        "applicant",    # 申请人本人
        "initiator"     # 流程发起人
    ]
    assignee_id: Optional[str] = None          # 固定user_id或role_id
    department_field: Optional[str] = None    # 部门字段来源 (e.g., "applicant.dept")
    fallback_assignee: Optional[str] = None   # 备选责任人
    escalation_rules: List[EscalationRule] = []  # 升级规则
    
    # ==================== 时限要求 ====================
    timeout: Optional[int] = None              # 超时时间(小时)
    timeout_unit: Literal["hours", "days"] = "hours"
    timeout_action: Literal[
        "auto_approve",    # 自动通过
        "auto_reject",    # 自动驳回
        "escalate",       # 升级
        "notify"          # 仅通知
    ] = "escalate"
    
    # ==================== 通知告警 ====================
    notifications: NotificationConfig
    alerts: AlertConfig
    
    # ==================== 输入输出 ====================
    inputs: List[InputSource]                 # 输入来源
    outputs: List[OutputTarget]                # 输出目标
    
    # ==================== 文档处理 ====================
    document_handling: Optional[DocumentConfig] = None
    
    # ==================== 表单配置 ====================
    form_config: Optional[FormConfig] = None   # 人工节点需要填写
    
    # ==================== 业务规则 ====================
    conditions: List[ConditionRule] = []      # 进入条件
    validators: List[ValidatorRule] = []       # 数据校验规则


class NotificationConfig(BaseModel):
    """通知配置"""
    
    # 通知触发时机
    on_entry: List[str] = []           # 进入节点时通知 ["manager", "applicant"]
    on_complete: List[str] = []         # 完成时通知
    on_timeout: List[str] = []         # 超时时通知
    on_escalate: List[str] = []        # 升级时通知
    
    # 催办提醒
    reminder: Optional[ReminderConfig] = None
    
    # 通知渠道
    channels: List[Literal["wecom", "email", "sms", "system"]] = ["wecom"]
    
    # 抄送配置
    cc: Optional[List[str]] = None


class ReminderConfig(BaseModel):
    """催办配置"""
```

```python
class InputSource(BaseModel):
    """输入来源定义"""
    
    source_type: Literal[
        "upstream",    # 上游节点
        "external",     # 外部系统触发
        "trigger",      # 触发器(定时/事件)
        "document",     # 文档导入
        "form",         # 表单填写
        "system"        # 系统生成
    ]
    source_id: Optional[str] = None     # 上游节点ID或外部系统标识
    field_mapping: Dict[str, str] = {} # 字段映射: {"目标字段": "来源字段"}
    required: bool = False             # 是否必需
    default_value: Optional[Any] = None # 默认值


class OutputTarget(BaseModel):
    """输出目标定义"""
    
    target_type: Literal[
        "downstream",   # 下游节点
        "external",     # 外部系统
        "archive",       # 归档
        "notification"   # 通知
    ]
    target_id: Optional[str] = None     # 下游节点ID或外部系统
    field_mapping: Dict[str, str] = {} # 字段映射
    condition: Optional[str] = None    # 输出条件 (e.g., "status == 'approved'")
    on_error: Literal["skip", "retry", "halt"] = "skip"


class ConditionRule(BaseModel):
    """条件规则"""
    field: str                         # 字段名
    operator: Literal["eq", "ne", "gt", "lt", "gte", "lte", "in", "not_in", "contains"]
    value: Any                        # 比较值
    error_message: Optional[str] = None # 不满足时的提示


class ValidatorRule(BaseModel):
    """数据校验规则"""
    validator_type: Literal["required", "format", "range", "custom"]
    field: str
    params: Dict[str, Any] = {}
    error_message: Optional[str] = None
```

**动态计算示例:**

```python
# 动态责任人计算示例
def resolve_dynamic_assignee(assignee_type: str, context: dict) -> str:
    """动态计算责任人"""
    
    if assignee_type == "applicant_direct_manager":
        # 获取申请人的直属上级
        user_id = context.get("applicant_id")
        user_info = get_user_info(user_id)
        return user_info.get("manager_id")
    
    elif assignee_type == "department_head":
        # 部门负责人
        dept_id = context.get("applicant_department")
        dept_info = get_department_info(dept_id)
        return dept_info.get("head_id")
    
    elif assignee_type == "next_approver":
        # 按审批顺序下一级
        current_level = context.get("current_approval_level")
        return get_next_approver(current_level)
    
    else:
        raise ValueError(f"Unknown assignee_type: {assignee_type}")
```

```python
class DataArchiveSpec(BaseModel):
    """数据归档规范"""
    
    # 归档类型
    archive_type: Literal["process", "document", "attachment", "log", "analysis"]

    # 向量化策略 (用于知识库)
    vectorization: Optional[VectorizationPolicy] = None
    
    # 归档格式
    format: Literal["json", "pdf", "markdown", "sql"] = "json"

    """归档内容定义"""
    source: str                          # 来源: "node_output", "form_data", "system"
    field: str                           # 字段名
    description: Optional[str] = None    # 描述


class StorageConfig(BaseModel):
    """存储配置"""
    storage_type: Literal["database", "file_system", "object_storage"]
    location: str                       # 存储路径/表名
    
    # 压缩加密
    compress: bool = False
    encrypt: bool = False


class RetentionPolicy(BaseModel):
    """保留策略"""
    hot_storage_days: int = 30          # 热存储(SSD/高速存储)
    cold_storage_days: int = 365        # 冷存储
    archive_days: int = 2555            # 归档(7年)
    delete_after: Optional[str] = None  # 何时删除


class VectorizationPolicy(BaseModel):
    """向量化策略"""
    enabled: bool = True
    when: Literal["on_complete", "scheduled", "manual"] = "on_complete"
```

```sql
    completed_at TIMESTAMP,
    timeout_at TIMESTAMP
```

```python
# 伪代码：模板 → LangGraph 代码生成

def generate_langgraph_code(workflow_definition: dict) -> str:
    """根据工作流定义生成LangGraph代码"""
    
    nodes = workflow_definition.get("nodes", [])
    edges = workflow_definition.get("edges", [])
    
    code_parts = []
    
    # 1. 导入
    code_parts.append("""
from langgraph.graph import StateGraph, START, END
from typing import TypedDict

class WorkflowState(TypedDict):
    messages: list
    variables: dict
    current_node: str
""")
    
    # 2. 节点函数
    for node in nodes:
        code_parts.append(generate_node_function(node))
    
    # 3. 构建图
    code_parts.append("""
def create_workflow_graph():
    graph = StateGraph(WorkflowState)
    
    # 添加节点
""")
    for node in nodes:
        code_parts.append(f'    graph.add_node("{node["node_id"]}", {node["node_id"]}_node)')
    
    # 4. 添加边
    code_parts.append("""
    # 添加边
    graph.add_edge(START, "{first_node}")
""".format(first_node=edges[0]["source"] if edges else ""))
    
    for edge in edges:
        if edge.get("condition"):
            code_parts.append(f'''
    graph.add_conditional_edges(
        "{edge["source"]}", 
        {edge["source"]}_router,
        {{
            {format_condition_mapping(edge["condition"])}
        }}
    )
''')
        else:
            code_parts.append(f'    graph.add_edge("{edge["source"]}", "{edge["target"]}")')
    
    code_parts.append("    return graph.compile()")
    
    return "\n".join(code_parts)
```

### 11.5 微流程功能模块（货架管理）

> **重要说明**：微流程是我们架构设计的抽象概念，实际实现时是**功能模块/组件**，而非"微流程引擎"。配置工作流时，只需勾选需要的功能模块并填写参数即可，无需任何流程编排。

#### 11.5.1 设计原则

```
┌─────────────────────────────────────────────────────────────────────┐
│                    微流程 = 功能模块 (货架组件)                      │
├─────────────────────────────────────────────────────────────────────┤
│                                                                      │
│  错误理解:  工作流引擎 ──► 节点 ──► 微流程引擎 ──► 子流程         │
│            (搞了一套嵌套的流程系统，画虎不成反类犬)                  │
│                                                                      │
│  正确理解:  节点配置 = 功能模块开关 + 参数配置                      │
│                                                                      │
│  就像汽车配置表:                                                    │
│  ├── ✓ 倒车影像    ←─── 勾选即启用，不选就没有                      │
│  ├── ✓ 座椅加热    ←─── 勾选即启用，不选就没有                      │
│  └── ✓ 自动泊车    ←─── 勾选即启用，不选就没有                      │
│                                                                      │
└─────────────────────────────────────────────────────────────────────┘
```

```sql
-- 绩效评估记录表
CREATE TABLE performance_evaluations (
    record_id VARCHAR(50) PRIMARY KEY,
    evaluation_type VARCHAR(20) NOT NULL,  -- node/workflow/person
    
    -- 评估对象
    target_id VARCHAR(50) NOT NULL,
    target_type VARCHAR(20),
    target_name VARCHAR(100),
    
    -- 时间范围
    period_type VARCHAR(20),               -- daily/weekly/monthly
    period_start TIMESTAMP NOT NULL,
    period_end TIMESTAMP NOT NULL,
    
    -- 指标得分(JSONB)
    metrics JSONB NOT NULL,
    
    -- 综合得分
    total_score DECIMAL(5,2) NOT NULL,
    
    -- 评级
    rating VARCHAR(1),                      -- S/A/B/C/D
    
    -- 排名
    rank INT,
    total_count INT,
    
    -- 趋势
    trend VARCHAR(10),                      -- up/stable/down
    trend_rate DECIMAL(5,2),
    
    -- AI分析
    ai_analysis TEXT,
    
    -- 源数据引用
    source_data_ref VARCHAR(100),
    
    created_at TIMESTAMP DEFAULT NOW(),
    
    UNIQUE(target_id, evaluation_type, period_type, period_start)
);

-- 评估指标配置表
CREATE TABLE evaluation_metrics (
    metric_id VARCHAR(50) PRIMARY KEY,
    metric_name VARCHAR(100) NOT NULL,
    metric_type VARCHAR(20),               -- time/quality/compliance
    
    -- 数据来源
    source_table VARCHAR(50),
    source_field VARCHAR(50),
    calculation_formula TEXT,
    
    -- 计算权重
    weight DECIMAL(5,2) DEFAULT 1.0,
    
    -- 评级标准
    rating_thresholds JSONB,               -- {"S": 90, "A": 80, "B": 70, "C": 60}
    
    -- BSC映射
    bsc_perspective VARCHAR(50),
    bsc_objective VARCHAR(100),
    
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW()
);
```
